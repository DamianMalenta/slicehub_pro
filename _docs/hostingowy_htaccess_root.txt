# =============================================================================
# SliceHub — production .htaccess (uti.pl, OVH, h88, inne shared hostingi)
# -----------------------------------------------------------------------------
# Wgranie: menedżer plików hostingu → public_html/.htaccess
#          (root domeny, NIE głębiej!)
#
# CO MUSISZ UZUPEŁNIĆ (sekcja [1]):
#   - SLICEHUB_DB_NAME  ← nazwa Twojej bazy z panelu hostingu
#   - SLICEHUB_DB_USER  ← użytkownik bazy
#   - SLICEHUB_DB_PASS  ← hasło bazy
#   - JWT_SECRET        ← losowy ciąg min. 32 znaki (np. openssl rand -hex 32)
#
# OSOBNO STWÓRZ PLIK `slicehub.net/core/local_secrets.php` z drugim kluczem
# SLICEHUB_SCRIPT_KEY — szablon w pliku `local_secrets_template.php` w tym
# samym folderze _docs/.
#
# Wszystkie inne sekcje [2-9] zostaw bez zmian — są gotowe.
# =============================================================================


# =============================================================================
# [1] DB CREDENTIALS + JWT — wstrzykiwane do PHP-FPM przez SetEnv
# -----------------------------------------------------------------------------
# Czytane przez core/db_config.php (PDO connection + define JWT_SECRET).
# Bez nich PHP fallbackuje na XAMPP defaulty (root / '') → 1045 Access denied.
# =============================================================================

SetEnv SLICEHUB_DB_HOST localhost
SetEnv SLICEHUB_DB_NAME <TUTAJ_NAZWA_BAZY>
SetEnv SLICEHUB_DB_USER <TUTAJ_USER_BAZY>
SetEnv SLICEHUB_DB_PASS <TUTAJ_HASLO_BAZY>
SetEnv JWT_SECRET <TUTAJ_LOSOWY_CIAG_64_ZNAKI_HEX>

# OPCJONALNIE: SLICEHUB_TENANT_ID — wymuś konkretny tenant dla tego hostingu.
# Domyślnie tenant_config.php sam wykrywa pierwszego aktywnego tenanta z bazy
# (auto-discovery). Ustaw poniższe TYLKO jeśli na jednej domenie hostujesz wielu
# tenantów i chcesz forsować konkretnego, albo jeśli auto-discovery zwraca nie
# tego, którego chcesz. Wartość = ID z tabeli sh_tenant (SELECT id, name FROM sh_tenant).
# SetEnv SLICEHUB_TENANT_ID 2


# =============================================================================
# [2] REWRITE — Authorization header + alias /slicehub/
# =============================================================================

RewriteEngine On

# (2a) Apache zwykle obcina nagłówek Authorization. Bez tego JWT z aplikacji
#      mobilnych (Kelner / Kierowca / Driver App) nie dochodzi do PHP.
RewriteCond %{HTTP:Authorization} .
RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

# (2b) Fallback dla CGI/FastCGI setups
SetEnvIf Authorization "(.*)" HTTP_AUTHORIZATION=$1

# (2c) ALIAS /slicehub/X → /X
#      Kod ma w wielu miejscach absolutne ścieżki "/slicehub/..." (Service
#      Workery POS i Online, manifesty PWA, część wrapperów API). Lokalnie
#      (XAMPP) działa, bo projekt jest w htdocs/slicehub/. Na uti.pl pliki
#      są w public_html/ — bez tego aliasu API zwracałoby 404.
#      Reguła nie dotyczy localhost (XAMPP nieruszony).
RewriteCond %{HTTP_HOST} !^(localhost|127\.0\.0\.1) [NC]
RewriteRule ^slicehub/(.*)$ /$1 [L]


# =============================================================================
# [3] DEFAULT DOC + STRICT MIME
# =============================================================================

DirectoryIndex index.html index.php
AddDefaultCharset UTF-8

AddType application/javascript .js
AddType application/json .json
AddType application/manifest+json .webmanifest


# =============================================================================
# [4] BEZPIECZEŃSTWO — blokady plików
# =============================================================================

# Wszystkie dotfiles (.git, .env, .htpasswd itd.)
<FilesMatch "^\.">
    Require all denied
</FilesMatch>

# Pliki z sekretami / backupy
<FilesMatch "(^|/)(local_secrets\.php|db_config\.php|composer\.(json|lock)|package(-lock)?\.json|\.env(\..*)?|.*\.sql|.*\.bak|.*\.backup|.*\.old|.*\.tmp)$">
    Require all denied
</FilesMatch>

# Foldery służbowe (zwracaj 404 bez wycieku informacji)
RedirectMatch 404 /\.git(/|$)
RedirectMatch 404 /\.idea(/|$)
RedirectMatch 404 /\.vscode(/|$)
RedirectMatch 404 /node_modules(/|$)
RedirectMatch 404 /tests(/|$)
RedirectMatch 404 /_KOPALNIA_WIEDZY_LEGACY(/|$)
RedirectMatch 404 /_docs(/|$)
RedirectMatch 404 /database(/|$)

# Brak directory listing
Options -Indexes
IndexIgnore *

# Ukryj sygnaturę serwera
ServerSignature Off
<IfModule mod_headers.c>
    Header unset X-Powered-By
    Header always unset X-Powered-By
</IfModule>


# =============================================================================
# [5] SECURITY HEADERS — defense in depth
# =============================================================================

<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=86400; includeSubDomains"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
    Header always set Permissions-Policy "geolocation=(self), microphone=(), camera=()"
    Header set Service-Worker-Allowed "/"
</IfModule>


# =============================================================================
# [6] CACHE HINTS — assety vs API
# =============================================================================

<IfModule mod_expires.c>
    ExpiresActive On

    ExpiresByType application/json                  "access plus 0 seconds"
    ExpiresByType text/html                         "access plus 0 seconds"
    ExpiresByType application/manifest+json         "access plus 1 hours"

    ExpiresByType application/javascript            "access plus 1 hours"
    ExpiresByType text/css                          "access plus 1 hours"

    ExpiresByType image/jpeg                        "access plus 30 days"
    ExpiresByType image/png                         "access plus 30 days"
    ExpiresByType image/webp                        "access plus 30 days"
    ExpiresByType image/svg+xml                     "access plus 30 days"
    ExpiresByType font/woff2                        "access plus 90 days"
</IfModule>

# Service Workery muszą być zawsze świeże (krytyczne dla PWA POS/Online)
<FilesMatch "sw\.js$">
    <IfModule mod_headers.c>
        Header set Cache-Control "no-store, no-cache, must-revalidate, max-age=0"
        Header set Pragma "no-cache"
    </IfModule>
</FilesMatch>


# =============================================================================
# [7] KOMPRESJA
# =============================================================================

<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html
    AddOutputFilterByType DEFLATE text/css
    AddOutputFilterByType DEFLATE text/plain
    AddOutputFilterByType DEFLATE application/javascript
    AddOutputFilterByType DEFLATE application/json
    AddOutputFilterByType DEFLATE application/xml
    AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>


# =============================================================================
# [8] PHP — limity dla uploadów (zdjęcia menu, faktury PDF, zdjęcia dań)
# =============================================================================

<IfModule mod_php.c>
    php_value upload_max_filesize 16M
    php_value post_max_size       18M
    php_value max_execution_time  90
    php_value memory_limit        256M
</IfModule>
<IfModule mod_php7.c>
    php_value upload_max_filesize 16M
    php_value post_max_size       18M
    php_value max_execution_time  90
    php_value memory_limit        256M
</IfModule>
<IfModule mod_php8.c>
    php_value upload_max_filesize 16M
    php_value post_max_size       18M
    php_value max_execution_time  90
    php_value memory_limit        256M
</IfModule>


# =============================================================================
# [9] BŁĘDY
# =============================================================================

ErrorDocument 404 /index.html
